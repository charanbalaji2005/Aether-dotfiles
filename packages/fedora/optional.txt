# "complete" mode only. install.sh checks each name against your enabled
# repositories first, so a package a given Fedora release doesn't ship is
# reported and skipped instead of failing the whole install. Every feature below hides itself or reports "unavailable"
# when its package is missing.

glx-utils               # glxinfo (XWayland fallback)
mesa-vulkan-drivers     # RADV / ANV Vulkan drivers
